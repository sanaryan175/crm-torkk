IN YOUR CODE

import { useTheme } from '@/lib/theme-context';

function MyComponent() {
  const { theme, isDark, setTheme } = useTheme();
  
  return (
    <div>
      Current: {theme}
      {isDark && <p>Dark mode enabled</p>}
      <button onClick={() => setTheme('dark')}>
        Enable Dark
      </button>
    </div>
  );
}
